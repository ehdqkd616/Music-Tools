# Lead/backing vocal split — isolated CPU-only image (see requirements-mdx.txt
# for why it can't share worker-separate's environment).
FROM python:3.11-slim

WORKDIR /srv

RUN apt-get update && apt-get install -y --no-install-recommends \
    ffmpeg libsndfile1 gcc g++ \
    && rm -rf /var/lib/apt/lists/*

COPY services/workers/requirements-mdx.txt ./requirements-mdx.txt
RUN pip install --no-cache-dir -r requirements-mdx.txt

COPY services/common ./common
# The whole tree, because workers/tasks/__init__.py imports every task module.
COPY services/workers ./workers

ENV PYTHONPATH=/srv PYTHONUNBUFFERED=1

CMD ["celery", "-A", "workers.celery_app", "worker", "-Q", "mdx", "-c", "1", "-P", "solo", "--loglevel=INFO"]
